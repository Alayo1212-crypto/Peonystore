  const [enableCustomization, setEnableCustomization] = useState(false);

  const [orderSuccess, setOrderSuccess] = useState(false);

  // Form State for Admin Panel Add/Edit

    nameEn: '',
    category: 'smart-pens',
    price: '',
    descriptionAr: '',
    descriptionEn: '',
    image: '',
    allowCustomization: true,
    inStock: true

    localStorage.setItem('peony_products', JSON.stringify(products));
  }, [products]);

  // Filtered products list matching only categories allowed
  const filteredProducts = products.filter(product => {
    const matchesCategory = selectedCategory === 'all' || product.category === selectedCategory;
    const matchesSearch = 
      product.nameAr.toLowerCase().includes(searchQuery.toLowerCase()) ||
      product.nameEn.toLowerCase().includes(searchQuery.toLowerCase()) ||
      product.descriptionAr.toLowerCase().includes(searchQuery.toLowerCase());

    setCart(prev => {
      const itemKey = product.id + (customNote ? `_custom_${customNote}` : '');
      const existing = prev.find(item => item.cartItemId === itemKey);
      if (existing) {
        return prev.map(item => item.cartItemId === itemKey ? { ...item, quantity: item.quantity + 1 } : item);
      }
      return [...prev, { ...product, cartItemId: itemKey, customNote, quantity: 1 }];
    });
    setIsCartOpen(true);
  };

    setCart(prev => prev.map(item => {
      if (item.cartItemId === cartItemId) {
        const newQty = item.quantity + delta;
        return newQty > 0 ? { ...item, quantity: newQty } : null;
      }
      return item;
    }).filter(Boolean));
  };

    setCart(prev => prev.filter(item => item.cartItemId !== cartItemId));

    setCart(prev => prev.map(item => {
      if (item.id === id) {
        const newQty = item.quantity + delta;
        return newQty > 0 ? { ...item, quantity: newQty } : null;
      }
      return item;
    }).filter(Boolean));
  };

  const removeFromCart = (id) => {
    setCart(prev => prev.filter(item => item.id !== id));

    e.preventDefault();

      setProducts(prev => prev.map(p => p.id === editingProductId ? {
        ...p,
        ...productForm,
        price: parseFloat(productForm.price)
      } : p));

      const newProduct = {
        id: 'p_' + Date.now(),
        ...productForm,
        price: parseFloat(productForm.price)
      };
      setProducts(prev => [newProduct, ...prev]);

    resetForm();

      nameAr: '',
      nameEn: '',
      category: 'smart-pens',
      price: '',
      descriptionAr: '',
      descriptionEn: '',
      image: '',
      inStock: true

      nameAr: product.nameAr,
      nameEn: product.nameEn,
      category: product.category,
      price: product.price,
      descriptionAr: product.descriptionAr,
      descriptionEn: product.descriptionEn,
      image: product.image,
      inStock: product.inStock

              </button>

          </div>
        </div>
      </header>

      {/* HERO / BRAND BANNER */}
      <section className="relative overflow-hidden bg-gradient-to-b from-pink-50/60 to-white py-10 px-4 sm:px-6">
        <div className="max-w-5xl mx-auto text-center space-y-3">
          <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full bg-white border border-pink-200 shadow-xs text-xs text-pink-700">
            <span className="w-2 h-2 rounded-full bg-pink-500 animate-ping"></span>
            {lang === 'ar' ? 'متجر رسمي معتمد' : 'Official Verified Boutique'}
          </div>
          <h1 className="text-3xl sm:text-4xl font-serif font-bold text-slate-900">
            {lang === 'ar' ? 'تشكيلة بيوني الفاخرة' : 'PEONY Luxury Collection'}

        {/* Category Header Title */}

            <span>
              {selectedCategory === 'all'

              onClick={() => setIsAdminOpen(true)}
              className="px-5 py-2.5 bg-slate-900 text-white rounded-xl text-xs font-semibold hover:bg-slate-800 transition shadow-sm inline-flex items-center gap-2"

              <Plus className="w-4 h-4" />
              <span>{lang === 'ar' ? 'إضافة منتج للأدمن' : 'Add Admin Product'}</span>

          /* Grid of Products */

            {filteredProducts.map(product => {
              const categoryObj = CATEGORIES.find(c => c.id === product.category);

                <div 
                  key={product.id}
                  className="group bg-white rounded-2xl border border-slate-100 shadow-xs hover:shadow-md transition-all duration-300 flex flex-col overflow-hidden"
                >
                  {/* Product Image */}

                    <img

                      <span>{categoryObj?.icon}</span>
                      <span>{lang === 'ar' ? categoryObj?.nameAr : categoryObj?.nameEn}</span>
                    </div>
                  </div>

                  {/* Details */}
                  <div className="p-4 flex-1 flex flex-col justify-between">
                    <div>
                      <h3 className="font-bold text-slate-800 text-sm mb-1 line-clamp-1 group-hover:text-pink-600 transition-colors">
                        {lang === 'ar' ? product.nameAr : product.nameEn}
                      </h3>
                      <p className="text-xs text-slate-500 line-clamp-2 mb-3 leading-relaxed">
                        {lang === 'ar' ? product.descriptionAr : product.descriptionEn}
                      </p>

                      <div className="flex flex-col">
                        <span className="text-[10px] text-slate-400">{lang === 'ar' ? 'السعر' : 'Price'}</span>

        <div className="max-w-7xl mx-auto px-4 py-12 sm:px-6 lg:px-8">
          <div className="grid grid-cols-1 md:grid-cols-3 gap-8 mb-8">
            <div>
              <div className="flex items-center gap-3 mb-3">
                <div className="w-8 h-8 rounded-full overflow-hidden border border-pink-300 bg-white">

            <div className="p-4 border-b border-slate-100 flex items-center justify-between bg-slate-50">
              <div className="flex items-center gap-2">
                <ShoppingBag className="w-5 h-5 text-pink-500" />
                <h3 className="font-bold text-slate-800 text-base">{lang === 'ar' ? 'عربة التسوق' : 'Shopping Cart'}</h3>
              </div>
              <button onClick={() => { setIsCartOpen(false); setCheckoutStep(false); }} className="p-1 rounded-full hover:bg-slate-200">
                <X className="w-5 h-5 text-slate-500" />
              </button>
            </div>

            {/* Order Success Overlay */}
            {orderSuccess ? (
              <div className="flex-1 p-8 text-center flex flex-col items-center justify-center space-y-4">
                <div className="w-16 h-16 bg-green-100 rounded-full flex items-center justify-center text-green-600 mb-2 animate-bounce">
                  <Check className="w-8 h-8" />
                </div>
                <h3 className="text-xl font-bold text-slate-800">{lang === 'ar' ? 'تم استلام طلبك بنجاح!' : 'Order Placed Successfully!'}</h3>
                <p className="text-xs text-slate-500 leading-relaxed">
                  {lang === 'ar' ? 'شكراً لتسوقك من PEONY BOUTIQUE. سنقوم بالتواصل معك قريباً لـتأكيد الشحن.' : 'Thank you for shopping at PEONY BOUTIQUE.'}
                </p>
              </div>
            ) : checkoutStep ? (
              /* Checkout Form */
              <form onSubmit={handleCheckoutSubmit} className="flex-1 overflow-y-auto p-5 space-y-4">
                <h4 className="font-bold text-sm text-slate-800 border-b pb-2">{lang === 'ar' ? 'تفاصيل الشحن والطلب' : 'Shipping Details'}</h4>
                
                <div>
                  <label className="block text-xs font-medium text-slate-600 mb-1">{lang === 'ar' ? 'الاسم الكامل' : 'Full Name'}</label>
                  <input required type="text" className="w-full text-xs p-2.5 border border-slate-200 rounded-lg focus:ring-2 focus:ring-pink-300 outline-none" />
                </div>

                <div>
                  <label className="block text-xs font-medium text-slate-600 mb-1">{lang === 'ar' ? 'رقم الهاتف' : 'Phone Number'}</label>
                  <input required type="tel" className="w-full text-xs p-2.5 border border-slate-200 rounded-lg focus:ring-2 focus:ring-pink-300 outline-none" />
                </div>

                <div>
                  <label className="block text-xs font-medium text-slate-600 mb-1">{lang === 'ar' ? 'عنوان التوصيل' : 'Delivery Address'}</label>
                  <textarea required rows={2} className="w-full text-xs p-2.5 border border-slate-200 rounded-lg focus:ring-2 focus:ring-pink-300 outline-none"></textarea>
                </div>

                <div className="bg-pink-50 p-3 rounded-xl border border-pink-100 mt-4">
                  <div className="flex justify-between text-xs font-bold text-slate-800 mb-1">
                    <span>{lang === 'ar' ? 'إجمالي السعر:' : 'Total Amount:'}</span>
                    <span>{cartTotal} د.ل</span>
                  </div>
                  <span className="text-[10px] text-pink-700">{lang === 'ar' ? 'الدفع عند الاستلام متاحة' : 'Cash on delivery available'}</span>
                </div>

                <div className="flex gap-2 pt-4">
                  <button type="button" onClick={() => setCheckoutStep(false)} className="flex-1 py-2.5 border border-slate-300 rounded-xl text-xs font-semibold hover:bg-slate-50">
                    {lang === 'ar' ? 'الرجوع للعربة' : 'Back to Cart'}
                  </button>
                  <button type="submit" className="flex-1 py-2.5 bg-pink-500 text-white rounded-xl text-xs font-semibold hover:bg-pink-600 shadow-sm">
                    {lang === 'ar' ? 'تأكيد الطلب الان' : 'Confirm Order'}
                  </button>
                </div>
              </form>
            ) : (
              /* Items List */
              <>
                <div className="flex-1 overflow-y-auto p-4 space-y-3">
                  {cart.length === 0 ? (
                    <div className="h-full flex flex-col items-center justify-center text-center text-slate-400 space-y-2">
                      <ShoppingBag className="w-12 h-12 opacity-30" />
                      <p className="text-xs">{lang === 'ar' ? 'عربة التسوق فارغة' : 'Your cart is empty'}</p>
                    </div>
                  ) : (
                    cart.map(item => (
                      <div key={item.cartItemId} className="flex gap-3 p-2.5 bg-slate-50 rounded-xl border border-slate-100 items-center">
                        <img src={item.image} alt={item.nameAr} className="w-14 h-14 object-cover rounded-lg bg-white" />
                        <div className="flex-1 min-w-0">
                          <h4 className="text-xs font-bold text-slate-800 truncate">{lang === 'ar' ? item.nameAr : item.nameEn}</h4>
                          {item.customNote && (
                            <span className="inline-block bg-pink-100 text-pink-700 text-[10px] px-2 py-0.5 rounded-md mt-0.5 border border-pink-200">
                              ✍️ {lang === 'ar' ? 'التخصيص:' : 'Custom:'} "{item.customNote}"
                            </span>
                          )}
                          <div className="text-xs font-bold text-pink-600 mt-0.5">{item.price} د.ل</div>
                        </div>
                        <div className="flex items-center gap-1.5 bg-white border border-slate-200 rounded-lg p-1">
                          <button onClick={() => updateCartQuantity(item.cartItemId, -1)} className="w-5 h-5 flex items-center justify-center text-slate-600 hover:bg-slate-100 rounded font-bold">-</button>
                          <span className="text-xs font-semibold px-1">{item.quantity}</span>
                          <button onClick={() => updateCartQuantity(item.cartItemId, 1)} className="w-5 h-5 flex items-center justify-center text-slate-600 hover:bg-slate-100 rounded font-bold">+</button>
                        </div>
                        <button onClick={() => removeFromCart(item.cartItemId)} className="p-1 text-slate-400 hover:text-red-500">
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </div>
                    ))
                  )}
                </div>

                {cart.length > 0 && (
                  <div className="p-4 border-t border-slate-100 bg-slate-50 space-y-3">
                    <div className="flex justify-between items-center text-sm font-bold text-slate-800">
                      <span>{lang === 'ar' ? 'الإجمالي:' : 'Total:'}</span>
                      <span className="text-pink-600 text-lg">{cartTotal} د.ل</span>
                    </div>
                    <button
                      onClick={() => setCheckoutStep(true)}
                      className="w-full py-3 bg-slate-900 text-white rounded-xl text-xs font-semibold hover:bg-slate-800 transition shadow-sm"
                    >
                      {lang === 'ar' ? 'متابعة الشراء' : 'Proceed to Checkout'}
                    </button>
                  </div>
                )}
              </>
            )}
          </div>
        </div>
      )}

      {/* ADMIN PANEL MODAL */}
      {isAdminOpen && (
        <div className="fixed inset-0 z-50 bg-black/50 backdrop-blur-xs flex items-center justify-center p-4 overflow-y-auto">
          <div className="bg-white rounded-2xl max-w-3xl w-full max-h-[90vh] overflow-y-auto shadow-2xl p-6 relative">
            
            {/* Close Admin */}
            <button 
              onClick={() => { setIsAdminOpen(false); resetForm(); }}
              className="absolute top-4 left-4 p-2 text-slate-400 hover:text-slate-600 rounded-full hover:bg-slate-100"
            >
              <X className="w-5 h-5" />
            </button>

            <div className="flex items-center gap-2 mb-6 border-b pb-3">
              <ShieldCheck className="w-6 h-6 text-pink-500" />
              <h2 className="text-lg font-bold text-slate-900">
                {lang === 'ar' ? 'لوحة إدارة متجر PEONY' : 'PEONY Store Admin Dashboard'}
              </h2>
            </div>

            {/* Strict Notice */}
            <div className="bg-amber-50 border border-amber-200 text-amber-800 text-xs p-3 rounded-xl mb-6 flex items-start gap-2">
              <AlertCircle className="w-4 h-4 text-amber-600 shrink-0 mt-0.5" />
              <span>
                {lang === 'ar' 
                  ? 'تنبيه إداري: المسموح فقط بتبويب وإضافة منتجات تحت الفئات الخمس الرسمية المحددة بالأسفل ولا يُسمح بالخروج عنها.'
                  : 'Strict restriction: Only products belonging to the 5 official categories can be created.'}
              </span>
            </div>

            {/* ADD / EDIT FORM */}
            <form onSubmit={handleSaveProduct} className="bg-slate-50 p-4 rounded-xl border border-slate-200 mb-8 space-y-4">
              <h3 className="font-bold text-sm text-slate-800">
                {editingProductId ? (lang === 'ar' ? 'تعديل المنتج الحالي' : 'Edit Product') : (lang === 'ar' ? 'إضافة منتج جديد' : 'Add New Product')}
              </h3>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-semibold text-slate-700 mb-1">{lang === 'ar' ? 'اسم المنتج (عربي)' : 'Product Name (AR)'}</label>
                  <input
                    required
                    type="text"
                    value={productForm.nameAr}
                    onChange={e => setProductForm({...productForm, nameAr: e.target.value})}
                    className="w-full text-xs p-2.5 bg-white border border-slate-300 rounded-lg outline-none focus:ring-2 focus:ring-pink-300"
                    placeholder="مثال: قلم لمس ذكي"
                  />
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-700 mb-1">{lang === 'ar' ? 'اسم المنتج (إنجليزي)' : 'Product Name (EN)'}</label>
                  <input
                    type="text"
                    value={productForm.nameEn}
                    onChange={e => setProductForm({...productForm, nameEn: e.target.value})}
                    className="w-full text-xs p-2.5 bg-white border border-slate-300 rounded-lg outline-none focus:ring-2 focus:ring-pink-300"
                    placeholder="e.g. Smart Pen"
                  />
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-700 mb-1">{lang === 'ar' ? 'الفئة (اختر من الفئات المعتمدة 5 فقط)' : 'Category (5 strictly allowed)'}</label>
                  <select
                    value={productForm.category}
                    onChange={e => setProductForm({...productForm, category: e.target.value})}
                    className="w-full text-xs p-2.5 bg-white border border-slate-300 rounded-lg outline-none focus:ring-2 focus:ring-pink-300 font-medium"
                  >
                    {CATEGORIES.map(c => (
                      <option key={c.id} value={c.id}>
                        {c.icon} {c.nameAr} - {c.nameEn}
                      </option>
                    ))}
                  </select>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-700 mb-1">{lang === 'ar' ? 'السعر (د.ل)' : 'Price (LYD)'}</label>
                  <input
                    required
                    type="number"
                    value={productForm.price}
                    onChange={e => setProductForm({...productForm, price: e.target.value})}
                    className="w-full text-xs p-2.5 bg-white border border-slate-300 rounded-lg outline-none focus:ring-2 focus:ring-pink-300"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1">{lang === 'ar' ? 'رابط صورة المنتج (URL)' : 'Product Image URL'}</label>
                <input
                  type="text"
                  value={productForm.image}
                  onChange={e => setProductForm({...productForm, image: e.target.value})}
                  className="w-full text-xs p-2.5 bg-white border border-slate-300 rounded-lg outline-none focus:ring-2 focus:ring-pink-300"
                  placeholder="https://..."
                />
              </div>

              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1">{lang === 'ar' ? 'وصف المنتج' : 'Description'}</label>
                <textarea
                  rows={2}
                  value={productForm.descriptionAr}
                  onChange={e => setProductForm({...productForm, descriptionAr: e.target.value})}
                  className="w-full text-xs p-2.5 bg-white border border-slate-300 rounded-lg outline-none focus:ring-2 focus:ring-pink-300"
                ></textarea>
              </div>

              <div className="flex gap-2 justify-end pt-2">
                {editingProductId && (
                  <button type="button" onClick={resetForm} className="px-4 py-2 border rounded-lg text-xs font-semibold">
                    {lang === 'ar' ? 'إلغاء' : 'Cancel'}
                  </button>
                )}
                <button type="submit" className="px-5 py-2 bg-pink-500 hover:bg-pink-600 text-white rounded-lg text-xs font-bold transition">
                  {editingProductId ? (lang === 'ar' ? 'حفظ التعديلات' : 'Save Changes') : (lang === 'ar' ? 'إضافة للمتجر' : 'Add Product')}
                </button>
              </div>
            </form>

            {/* EXISTING PRODUCTS LIST IN ADMIN */}
            <h3 className="font-bold text-sm text-slate-800 mb-3">{lang === 'ar' ? 'قائمة المنتجات الحالية' : 'Current Store Inventory'}</h3>
            <div className="space-y-2 max-h-60 overflow-y-auto pr-1">
      {/* QUICK VIEW & CUSTOMIZATION MODAL */}
      {selectedProductDetails && (
        <div className="fixed inset-0 z-50 bg-black/50 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white rounded-2xl max-w-lg w-full p-6 relative shadow-2xl">
            <button 
              onClick={() => { setSelectedProductDetails(null); setCustomText(''); setEnableCustomization(false); }}
              className="absolute top-4 left-4 p-1.5 text-slate-400 hover:text-slate-600 rounded-full hover:bg-slate-100"
            >
              <X className="w-5 h-5" />
            </button>

            <div className="flex flex-col sm:flex-row gap-4 mb-4">
              <img src={selectedProductDetails.image} alt="" className="w-full sm:w-44 h-44 object-cover rounded-xl bg-slate-100" />
              <div className="flex-1 space-y-2">
                <span className="text-[10px] bg-pink-100 text-pink-700 px-2.5 py-1 rounded-full font-bold">
                  {CATEGORIES.find(c => c.id === selectedProductDetails.category)?.nameAr}
                </span>
                <h3 className="font-bold text-slate-900 text-base">{lang === 'ar' ? selectedProductDetails.nameAr : selectedProductDetails.nameEn}</h3>
                <p className="text-xs text-slate-600 leading-relaxed">{lang === 'ar' ? selectedProductDetails.descriptionAr : selectedProductDetails.descriptionEn}</p>
                <div className="text-xl font-bold text-slate-900 pt-1">{selectedProductDetails.price} د.ل</div>
              </div>
            </div>

            {/* Customization Option Box */}
            <div className="bg-pink-50/70 border border-pink-200/80 rounded-xl p-3 mb-5 space-y-2">
              <div className="flex items-center justify-between">
                <label className="text-xs font-bold text-pink-900 flex items-center gap-1.5 cursor-pointer">
                  <input 
                    type="checkbox" 
                    checked={enableCustomization} 
                    onChange={(e) => setEnableCustomization(e.target.checked)}
                    className="rounded text-pink-500 focus:ring-pink-400"
                  />
                  <span>{lang === 'ar' ? 'طلب تخصيص (حفر اسم / كتابة إهداء)' : 'Customization (Name/Engraving)'}</span>
                </label>
                <Sparkles className="w-4 h-4 text-pink-500" />
              </div>

              {enableCustomization && (
                <input
                  type="text"
                  value={customText}
                  onChange={(e) => setCustomText(e.target.value)}
                  placeholder={lang === 'ar' ? 'اكتب الاسم أو النص المراد نقشه...' : 'Enter custom text or name...'}
                  className="w-full text-xs p-2.5 bg-white border border-pink-300 rounded-lg outline-none focus:ring-2 focus:ring-pink-400"
                />
              )}
            </div>

            <button
              onClick={() => {
                addToCart(selectedProductDetails, enableCustomization ? customText : '');
                setSelectedProductDetails(null);
                setCustomText('');
                setEnableCustomization(false);
              }}
              className="w-full py-2.5 bg-pink-500 hover:bg-pink-600 text-white rounded-xl text-xs font-bold transition flex items-center justify-center gap-2"
            >
              <ShoppingBag className="w-4 h-4" />
              <span>{lang === 'ar' ? 'إضافة إلى العربة' : 'Add to Cart'}</span>
            </button>
          </div>
        </div>
      )}

    </div>
  );
}
